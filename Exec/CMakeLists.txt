if(WIN32)
	add_executable(SailorExec WIN32 Main.cpp SailorEngine.manifest)
else()
	add_executable(SailorExec Main.cpp)
endif()
add_dependencies(SailorExec SailorLib)
target_link_libraries(SailorExec SailorLib)
target_compile_definitions(SailorExec PUBLIC NOMINMAX)
set_property(TARGET SailorExec PROPERTY FOLDER "Executables")
set_property(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR} PROPERTY VS_STARTUP_PROJECT SailorExec)
set_property(TARGET SailorExec PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${SAILOR_BINARIES_DIR}")
set_target_properties(SailorExec PROPERTIES OUTPUT_NAME "SailorEngine-$<CONFIG>")

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
	set_target_properties(SailorExec PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
elseif(APPLE)
	set_target_properties(SailorExec PROPERTIES INSTALL_RPATH "@executable_path/../${CMAKE_INSTALL_LIBDIR}")
endif()
install(TARGETS SailorExec RUNTIME DESTINATION bin COMPONENT Runtime)
if(WIN32)
	install(FILES $<TARGET_RUNTIME_DLLS:SailorExec> DESTINATION bin COMPONENT Runtime)
	if(SAILOR_BUILD_WITH_VULKAN)
		find_package(VulkanLoader CONFIG REQUIRED)
		install(FILES $<TARGET_FILE:Vulkan::Loader> DESTINATION bin COMPONENT Runtime)
	endif()
	if(MSVC)
		install(FILES $<TARGET_PDB_FILE:SailorExec> DESTINATION bin COMPONENT Runtime CONFIGURATIONS Debug OPTIONAL)
	endif()
endif()
